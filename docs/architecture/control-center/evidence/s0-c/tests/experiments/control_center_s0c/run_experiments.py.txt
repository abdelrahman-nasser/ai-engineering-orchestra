FAULTS = {
    'decode': dict(phase='whole_store_decode_validate', cancel=True, inside='decode'),
    'replay': dict(phase='indexed_replay_and_freeze', cancel=True, inside='validate_event'),
    'scheduler': dict(phase='core_scheduler', cancel=True, inside='subject_document'),
    'transaction': dict(phase='transaction_begin', cancel=True),
    'encoding': dict(phase='encoding', cancel=True, delay=0.03),
    'allocation': dict(phase='encoding', allocation_bytes=3*1024**2),
    'cleanup': dict(phase='cleanup', cancel=True, delay=0.03),
    'ownership-loss': dict(phase='indexed_replay_and_freeze', loss=True),
    'uncooperative-decode': dict(phase='whole_store_decode_validate', delay=1.05),
    'uncooperative-cleanup': dict(phase='cleanup', cancel=True, delay=1.05),
    'identity-mismatch': dict(phase='resource_preflight', expected_identity_wrong=True),
    'connection-failure': dict(phase='transaction_begin', closed_before_capture=True),
}


def worker(mode, shape, fault_name='none'):
    if mode not in ('timing', 'allocation', 'concurrent', 'fault'):
        raise ValueError('fixed S0-C worker mode required')
    if shape not in ('representative', 'long', 'alpha_max', 'core_max'):
        raise ValueError('fixed synthetic shape required')
    if fault_name != 'none' and fault_name not in FAULTS:
        raise ValueError('fixed fault required')
    trace_initial = tracemalloc.is_tracing()
    initial_memory = tracemalloc.get_traced_memory() if trace_initial else None
    if trace_initial: tracemalloc.stop()
    data = dict(mode=mode, shape=shape, fault=fault_name, python=sys.version,
        sqlite=sqlite3.sqlite_version, platform=platform.platform(),
        trace_initially_active=trace_initial, startup_trace_memory=initial_memory,
        native_sqlite=native_sqlite_status(),
        conditions='Fresh synthetic setup primes OS cache; not cold-disk or production-owner evidence.')
    with owned_scratch(os.environ.get('S0C_SCRATCH')) as root, ExitStack() as resources:
        admission = Admission()
        items = []
        number = 2 if mode == 'concurrent' else 1
        for index in range(number):
            directory = root / str(index); directory.mkdir()
            writer, reader, path, graph, fixture = establish(directory, shape)
            resources.callback(writer.close); resources.callback(reader.close)
            close_observations = []
            coordinator = Coordinator(close_callback=lambda: None)
            items.append((writer, reader, path, graph, fixture, coordinator))
        data['fixtures'] = [item[4] for item in items]
        data['process_before'] = process_memory()
        trace = mode in ('allocation', 'concurrent')
        if trace:
            gc.collect(); tracemalloc.start(1)
            baseline = tracemalloc.get_traced_memory(); tracemalloc.reset_peak()
        began = time.perf_counter_ns()
        if mode == 'concurrent':
            admissions_at_overlap = []
            barrier = threading.Barrier(2, action=lambda: admissions_at_overlap.append(asdict(admission.snapshot())))
            outputs, errors = [], []
            def run(index):
                try:
                    _, reader, path, graph, fixture, owner = items[index]
                    outputs.append(capture(reader, path, graph, admission, owner,
                        source=f's{index}', principal=f'p{index}', query_barrier=barrier,
                        expected_identity=tuple(fixture['setup_file_identity'])))
                except BaseException as error: errors.append(dict(type=type(error).__name__, message=str(error)))
            threads = [threading.Thread(target=run, args=(index,)) for index in range(2)]
            for thread in threads: thread.start()
            for thread in threads: thread.join(timeout=5)
            alive = [thread.name for thread in threads if thread.is_alive()]
            if alive:
                barrier.abort()
                for thread in threads: thread.join(timeout=3)
                if any(thread.is_alive() for thread in threads):
                    # Never unwind closers/delete scratch beneath a live user.
                    # This is only the owned measurement CHILD, never an owner.
                    try:
                        failed = write_receipt(dict(blocking_failure='synthetic threads did not drain',
                            live_threads=[thread.name for thread in threads if thread.is_alive()],
                            completed_cleanup_acknowledged=False,
                            classification='NO-GO; whole controlled child termination is not in-process cleanup proof'),
                            'unsafe-drain')
                        print('S0C_BLOCKING_RECEIPT=' + str(failed), flush=True)
                    finally:
                        os._exit(3)
            data.update(captures=outputs, thread_errors=errors,
                overlap_admissions=admissions_at_overlap, threads_joined=True,
                per_query_native_attribution='UNVERIFIED; shared-process trace cannot isolate native/thread ownership')
        else:
            _, reader, path, graph, fixture, owner = items[0]
            fault = FAULTS.get(fault_name)
            if fault and fault.get('closed_before_capture'): reader.close()
            expected = (0, 0) if fault and fault.get('expected_identity_wrong') else tuple(fixture['setup_file_identity'])
            data['captures'] = [capture(reader, path, graph, admission, owner,
                cancel=threading.Event(), fault=fault, expected_identity=expected)]
        data['elapsed_ns'] = time.perf_counter_ns() - began
        if trace:
            current, peak = tracemalloc.get_traced_memory()
            data['python_trace'] = dict(baseline_current_bytes=baseline[0], baseline_peak_bytes=baseline[1],
                current_after_queries_bytes=current, peak_bytes=peak,
                qualification='Complete attempted query/cancellation work including synthetic projection when reached. Timing under tracing is instrumentation-inflated; no per-query native attribution.')
            tracemalloc.stop()
        data['process_after_query'] = process_memory()
        data['admission_after'] = asdict(admission.snapshot())
        data['envelopes_after_query'] = [database_envelope(item[2]) for item in items]
        for _, _, _, _, _, owner in items:
            snap = owner.snapshot()
            if snap.readers or snap.writer: raise AssertionError('BLOCKING: live owner lease after acknowledgement')
        if admission.snapshot().active: raise AssertionError('BLOCKING: admission leaked')
        data['shutdown_after_capture'] = [asdict(item[5].shutdown()) for item in items]
        start = time.perf_counter_ns()
        for writer, reader, *_ in items: reader.close(); writer.close()
        data['fixture_handles_close_ns'] = time.perf_counter_ns() - start
        gc.collect()
        data['process_after_cleanup'] = process_memory()
        data['blocking_hold_failure'] = any(result['hold_seconds'] is not None and
            not result['hold_within_one_second'] for result in data['captures'])
        data['no_usable_prefix_on_failures'] = all(result['success'] or result['no_usable_prefix_on_error'] for result in data['captures'])
        data['cleanup_acknowledged_only_after_work'] = all(result['cleanup_acknowledged'] and
            not result['closed_under_active_work'] for result in data['captures'])
        data['fault_qualification'] = ('Injected function-entry gates/delays and finite synthetic work; no arbitrary OS-I/O preemption proof. '
            'Uncooperative >1s cases are intentional counterexamples, not P07 passes.')
        data['assertion_failures'] = []
        if mode == 'fault':
            expected_errors = {'decode': 'CoordinationCancelled', 'replay': 'CoordinationCancelled',
                'scheduler': 'CoordinationCancelled', 'transaction': 'CoordinationCancelled',
                'encoding': 'RuntimeError', 'allocation': 'ValueError', 'cleanup': 'CoordinationCancelled',
                'ownership-loss': 'OwnershipLost', 'uncooperative-decode': 'CaptureDeadlineExceeded',
                'uncooperative-cleanup': 'CoordinationCancelled', 'identity-mismatch': 'ResourceIdentityChanged',
                'connection-failure': 'ProgrammingError'}
            result = data['captures'][0]
            correct_error = result['error'] is not None and result['error']['type'] == expected_errors[fault_name]
            if result['success'] or result['observation'] is not None or not correct_error:
                data['assertion_failures'].append('fault did not fail closed with expected error')
            if fault_name.startswith('uncooperative') and not data['blocking_hold_failure']:
                data['assertion_failures'].append('counterexample did not record held-duration failure')
        # Each thread owns its resource; an admission snapshot can legitimately
        # show the other query still active. Final zero is checked above.
        if any(not item['resources_closed'] or item['closed_under_active_work'] or
               not (item['transaction_inactive_observed'] or item['connection_already_closed'])
               for item in data['captures']):
            data['assertion_failures'].append('cleanup observation failed')
    return data


def parent():
    evidence = dict(phase='S0-C', command=[sys.executable, '-B', str(Path(__file__).resolve())],
        cwd=str(REPOSITORY), child_timeout_seconds=60, serial=True, jobs=[],
        startup_pythontracemalloc=os.environ.get('PYTHONTRACEMALLOC'))
    started = time.perf_counter_ns()
    with owned_scratch() as root:
        environment = os.environ.copy()
        environment.update(S0C_SCRATCH=str(root), TEMP=str(root), TMP=str(root), PYTHONDONTWRITEBYTECODE='1')
        environment.pop('PYTHONTRACEMALLOC', None)
        base = [sys.executable, '-B']
        commands = [base + [str(Path(__file__).with_name(name))] for name in ('test_concurrency.py', 'test_resources.py')]
        for shape in ('representative', 'long', 'alpha_max', 'core_max'):
            commands.append(base + [str(Path(__file__).resolve()), '--worker', 'timing', shape, 'none'])
        for shape in ('representative', 'alpha_max'):
            commands.append(base + [str(Path(__file__).resolve()), '--worker', 'allocation', shape, 'none'])
        commands.append(base + [str(Path(__file__).resolve()), '--worker', 'concurrent', 'representative', 'none'])
        for fault in FAULTS:
            commands.append(base + [str(Path(__file__).resolve()), '--worker', 'fault', 'representative', fault])
        for command in commands:
            print('START ' + ' '.join(command[2:]), flush=True)
            start = time.perf_counter_ns()
            try:
                child = subprocess.run(command, cwd=REPOSITORY, env=environment,
                    text=True, capture_output=True, timeout=60)
                job = dict(command=command, cwd=str(REPOSITORY), exit_code=child.returncode,
                    wall_ns=time.perf_counter_ns()-start, stdout=child.stdout, stderr=child.stderr)
                if '--worker' in command and child.returncode == 0:
                    job['measurement'] = json.loads(child.stdout); del job['stdout']
            except subprocess.TimeoutExpired:
                job = dict(command=command, cwd=str(REPOSITORY), exit_code=None,
                    wall_ns=time.perf_counter_ns()-start, blocking_failure='child timed out/killed/reaped; no in-process cleanup proof')
            evidence['jobs'].append(job)
            print('END exit=' + str(job['exit_code']), flush=True)
            if job['exit_code'] != 0:
                evidence['stopped_on_failed_execution'] = True
                break
    evidence['wall_ns'] = time.perf_counter_ns() - started
    evidence['all_execution_jobs_exit_zero'] = len(evidence['jobs']) == len(commands) and all(job['exit_code'] == 0 for job in evidence['jobs'])
    evidence['source_hashes'] = {str(path.relative_to(REPOSITORY)): hashlib.sha256(path.read_bytes()).hexdigest()
        for path in Path(__file__).parent.glob('*.py')}
    evidence['feasibility_recommendation'] = 'NO-GO; full safety/native/P07 guarantees remain unverified; preserve hold-overrun counterexamples.'
    receipt = write_receipt(evidence)
    print('S0C_RECEIPT=' + str(receipt), flush=True)
    return 0 if evidence['all_execution_jobs_exit_zero'] else 1


if __name__ == '__main__':
    if len(sys.argv) == 5 and sys.argv[1] == '--worker':
        with (mock.patch.object(helpers.lifecycle_store.SqliteTaskRunStateStore, '__init__', side_effect=AssertionError('no owner')),
              mock.patch.object(helpers.lifecycle_store.SqliteTaskRunStateStore, 'provision', side_effect=AssertionError('no provisioning')),
              mock.patch.object(helpers.journal_store.SqliteExecutionJournalStore, '__init__', side_effect=AssertionError('no operational Journal'))):
            measured = worker(sys.argv[2], sys.argv[3], sys.argv[4])
            print(json.dumps(measured))
            sys.exit(1 if measured['assertion_failures'] else 0)
    elif len(sys.argv) == 1:
        sys.exit(parent())
    else: raise SystemExit('fixed S0-C arguments required')
