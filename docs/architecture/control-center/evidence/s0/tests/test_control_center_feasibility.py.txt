def fixture_roots():
    """Check all existing ancestors before any synthetic write; no junctions."""
    if not FIXTURES.is_relative_to(REPOSITORY) or FIXTURES.resolve() != FIXTURES:
        raise ValueError('fixture root is outside the authorized repository')
    for candidate in (CONTROLLED, *CONTROLLED.parents):
        if candidate.exists() or candidate.is_symlink():
            info = candidate.lstat()
            if candidate.is_symlink() or getattr(info, 'st_file_attributes', 0) & 0x400:
                raise ValueError('synthetic fixture ancestor is a reparse/symlink')
    if CONTROLLED.exists() and (not CONTROLLED.is_dir() or CONTROLLED.resolve() != CONTROLLED):
        raise ValueError('invalid controlled fixture directory')


def file_identity(path):
    info = path.lstat()
    return info.st_dev, info.st_ino


@contextmanager
def owned_directory():
    """Delete only a newly allocated, marked, contained synthetic directory."""
    fixture_roots()
    fixture_identity = file_identity(FIXTURES)
    CONTROLLED.mkdir(exist_ok=True)
    fixture_roots()
    controlled_identity = file_identity(CONTROLLED)
    root = Path(tempfile.mkdtemp(prefix='s0-owned-', dir=CONTROLLED))
    fixture_roots()
    if root.resolve() != root or root.is_symlink():
        raise AssertionError('allocated synthetic directory is not exact')
    root_identity = file_identity(root)
    marker = uuid.uuid4().hex
    (root / 'ownership-marker').write_text(marker, encoding='ascii')
    try:
        yield root
    finally:
        fixture_roots()
        if (file_identity(FIXTURES) != fixture_identity
                or file_identity(CONTROLLED) != controlled_identity
                or file_identity(root) != root_identity
                or root.parent != CONTROLLED.resolve() or not root.name.startswith('s0-owned-')
                or not root.is_relative_to(FIXTURES.resolve()) or root.is_symlink()
                or (root / 'ownership-marker').read_text(encoding='ascii') != marker):
            raise AssertionError('synthetic cleanup containment/ownership failure')
        for parent, directories, files in os.walk(root, followlinks=False):
            for name in (*directories, *files):
                item = Path(parent) / name
                info = item.lstat()
                if (item.is_symlink() or not item.resolve().is_relative_to(root)
                        or getattr(info, 'st_file_attributes', 0) & 0x400
                        or (item.is_file() and info.st_nlink != 1)):
                    raise AssertionError('synthetic cleanup rejected foreign object')
        fixture_roots()
        if file_identity(root) != root_identity:
            raise AssertionError('synthetic directory identity changed before removal')
        shutil.rmtree(root)

def record(subject=0, sequence=1, *, result=False, run_id=None):
    """Canonical-shaped descriptive fixture; its provenance is not authenticated."""
    contract = dict(task_id='synthetic-task', workflow_id='w', stage_id='s',
        role_id='r', actor_id='a', runtime_option_id='rt', option_id='o',
        environment_id='e', operation_id='repository_file_read',
        resource='synthetic.txt', execution_mode='standard')
    run = dict(run_id=f'r{subject}' if run_id is None else run_id, contract=contract)
    grant = dict(grant_id=f'g{subject}', run=run, authorization_domain_id='d',
        issuer_kind='human', issuer_id='i', provenance_reference='SYNTHETIC-NOT-AUTHORITY',
        issued_at='2026-10-17T00:00:00Z', expires_at='2026-10-18T00:00:00Z')
    return dict(format_version=1, journal_id='j', sequence=sequence,
        recorded_at=FIXED_TIME,
        ledger_identity=dict(authorization_domain_id='d', ledger_instance_id='l',
                             domain_generation=1),
        execution_entry=dict(identity=['d', 'human', 'i', f'g{subject}'],
            claim_id='1' * 64, executor_instance_id='2' * 64, lease_generation=1,

            entry_time_key=journal._time('2026-10-17T00:00:01.000000Z')),
        admission=dict(grant=grant, tool_binding=dict(run=copy.deepcopy(run), tool_id='tool'),
                       decision_time='2026-10-17T00:00:00.000000Z'),
        event_kind='result_observed' if result else 'entry_observed',
        provenance='original_controlled_invocation',
        result_summary=dict(outcome='succeeded', diagnostic_code=None, byte_count=0)
                       if result else None)


def make_journal(path, count, *, corrupt_last=False):
    """Create original-schema synthetic rows without provisioning an owner/store."""
    writer = sqlite3.connect(path, isolation_level=None)
    try:
        writer.execute('PRAGMA journal_mode=WAL')
        writer.execute('PRAGMA wal_autocheckpoint=0')
        statements, _ = journal_store._schema_resource()
        for statement in statements:
            writer.execute(statement)
        writer.execute('INSERT INTO journal_metadata VALUES(1,?,?,?,?,1,?)',
            ('j', 'd', 'l', '1', journal_store.RESOURCE_SHA256))
        writer.execute(f'PRAGMA application_id={journal_store.APPLICATION_ID}')
        writer.execute(f'PRAGMA user_version={journal_store.SCHEMA_VERSION}')
        total_bytes = 0
        writer.execute('BEGIN')
        for index in range(count):
            item = record(index, index + 1)
            if corrupt_last and index == count - 1:
                item['result_summary'] = {'unexpected': 'synthetic corruption'}
            encoded = journal._encode(item)
            total_bytes += len(encoded.encode('utf-8'))
            writer.execute('INSERT INTO journal_entries VALUES(?,?,?,?,?)',
                (index + 1, journal._encode(list(journal._subject_key(item))),
                 item['event_kind'], item['recorded_at'], encoded))
        writer.execute('COMMIT')
        config = journal_store.SqliteExecutionJournalConfiguration(
            path, path.parent, 'j', AuthorizationDomainIdentity('d', 'l', 1))
        return writer, SimpleNamespace(configuration=config), total_bytes
    except BaseException:
        writer.close()
        raise


def passive(path):
    connection = sqlite3.connect(path.as_uri() + '?mode=ro', uri=True,
        isolation_level=None, timeout=0.05)
    try:
        connection.row_factory = sqlite3.Row
        connection.execute('PRAGMA query_only=ON')
        return connection
    except BaseException:
        connection.close()
        raise


def scan(records, high_water, position, limit, cap=1000, run_id='r1500'):
    """Test-only traversal experiment, not an application query/cursor."""
    output, scanned = [], 0
    for item in records:
        document = item.document
        sequence = document['sequence']
        if sequence <= position or sequence > high_water:
            continue
        matched = document['admission']['grant']['run']['run_id'] == run_id
        if matched and len(output) == limit:
            break
        if scanned == cap:
            break
        position, scanned = sequence, scanned + 1
        if matched:
            output.append(sequence)
    return output, position, scanned, position == high_water


def process_memory():
    """Current process only; lifetime peaks are not per-query native allocation."""
    if os.name != 'nt':
        return {'supported': False}
    class Counters(ctypes.Structure):
        _fields_ = [('cb', wintypes.DWORD), ('PageFaultCount', wintypes.DWORD)] + [
            (name, ctypes.c_size_t) for name in ('PeakWorkingSetSize', 'WorkingSetSize',
            'QuotaPeakPagedPoolUsage', 'QuotaPagedPoolUsage', 'QuotaPeakNonPagedPoolUsage',
            'QuotaNonPagedPoolUsage', 'PagefileUsage', 'PeakPagefileUsage', 'PrivateUsage')]
    counter = Counters()
    counter.cb = ctypes.sizeof(counter)
    kernel = ctypes.WinDLL('kernel32', use_last_error=True)
    psapi = ctypes.WinDLL('psapi', use_last_error=True)
    kernel.GetCurrentProcess.restype = wintypes.HANDLE
    psapi.GetProcessMemoryInfo.argtypes = [wintypes.HANDLE, ctypes.c_void_p, wintypes.DWORD]
    if not psapi.GetProcessMemoryInfo(kernel.GetCurrentProcess(), ctypes.byref(counter), counter.cb):
        raise ctypes.WinError(ctypes.get_last_error())
    return {'working_set_bytes': counter.WorkingSetSize,
            'peak_working_set_bytes': counter.PeakWorkingSetSize,
            'private_bytes': counter.PrivateUsage,
            'peak_pagefile_bytes': counter.PeakPagefileUsage}


def lifecycle_documents(count):
    """One Task, repeated historical continuations; no real approval is produced."""
    events = [dict(format_version=1, sequence=1, task_id='synthetic-task', revision=1,
                   operation='enroll', data={'definition_reference': 'synthetic-definition'})]
    while len(events) < count:
        attempt = (len(events) - 1) // 3
        item = record(attempt)
        binding = item['admission']['tool_binding']
        operations = [
            ('register' if attempt == 0 else 'continue',
             {'subject': {'binding': binding, 'ledger_identity': item['ledger_identity']}}),
            ('start', {'run_id': binding['run']['run_id']}),
            ('result', {'run_id': binding['run']['run_id'], 'outcome': 'succeeded',
                        'result_identity': list(journal._subject_key(item))}),
        ]
        if attempt:
            operations[0][1].update(reason='synthetic-history',
                approval=dict(authority_kind='human', authority_id='synthetic',
                    session_nonce='synthetic-session', request_nonce='synthetic-request',
                    reference='SYNTHETIC-NOT-APPROVAL'))
        for operation, data in operations:
            if len(events) == count:
                break
            number = len(events) + 1
            events.append(dict(format_version=1, sequence=number, task_id='synthetic-task',
                               revision=number, operation=operation, data=data))
    return events


def measure(mode, count):
    if type(mode) is not str or mode not in ('journal', 'lifecycle'):
        raise ValueError('only journal/lifecycle synthetic measurements are allowed')
    if type(count) is not int or not 1 <= count <= 10000:
        raise ValueError('synthetic measurement count must be an exact integer in 1..10000')
    with owned_directory() as root, ExitStack() as resources:
        path = root / f'{mode}.sqlite3'
        if mode == 'journal':
            writer, probe, metadata_bytes = make_journal(path, count)
            resources.callback(writer.close)
        else:
            writer = sqlite3.connect(path, isolation_level=None)
            resources.callback(writer.close)
            writer.execute('PRAGMA journal_mode=DELETE')
            for _, _, statement in lifecycle_store._DDL:
                writer.execute(statement)
            writer.execute('INSERT INTO metadata VALUES(1,?,?,?)',
                           ('s', 'p', lifecycle_store._DDL_HASH))
            writer.execute(f'PRAGMA application_id={lifecycle_store.APPLICATION_ID}')
            writer.execute('PRAGMA user_version=1')
            documents = lifecycle_documents(count)
            payloads = [lifecycle.encode(item) for item in documents]
            metadata_bytes = sum(len(item.encode('utf-8')) for item in payloads)
            writer.execute('BEGIN')
            writer.executemany('INSERT INTO events VALUES(?,?)', enumerate(payloads, 1))
            writer.execute('COMMIT')
            del documents, payloads
            probe = SimpleNamespace(store_id='s', project_id='p')
        reader = passive(path)
        resources.callback(reader.close)
        if mode == 'lifecycle':
            reader.row_factory = None  # Existing lifecycle validator expects tuples.
        before = process_memory()
        statements = []
        reader.set_trace_callback(statements.append)
        try:
            reader.execute('BEGIN')
            start = time.perf_counter_ns()
            if mode == 'journal':
                value = journal_store.SqliteExecutionJournalStore._verify(probe, reader)
            else:
                probe._connection = reader
                value = lifecycle_store.SqliteTaskRunStateStore._verify(probe)
            elapsed = time.perf_counter_ns() - start
            reader.execute('ROLLBACK')
            after = process_memory()
            row_count = len(value[0])
            del value
            # A separate traced pass measures Python allocations, not native SQLite.
            traced_peak = None
            traced_elapsed = None
            if count <= APPROVED_P07.journal_records_per_request:
                tracemalloc.start()
                reader.execute('BEGIN')
                traced_start = time.perf_counter_ns()
                value = (journal_store.SqliteExecutionJournalStore._verify(probe, reader)
                         if mode == 'journal' else lifecycle_store.SqliteTaskRunStateStore._verify(probe))
                traced_elapsed = time.perf_counter_ns() - traced_start
                _, traced_peak = tracemalloc.get_traced_memory()
                tracemalloc.stop()
                del value
                reader.execute('ROLLBACK')
            sidecars = {suffix or 'database': Path(str(path) + suffix).stat().st_size
                        if Path(str(path) + suffix).exists() else None
                        for suffix in ('', '-wal', '-shm', '-journal')}
            return dict(mode=mode, records=row_count, source_metadata_bytes=metadata_bytes,
                validation_ns=elapsed, within_one_second=elapsed <= 1_000_000_000,
                within_ten_seconds=elapsed <= 10_000_000_000,
                python_tracemalloc_peak_bytes=traced_peak, traced_validation_ns=traced_elapsed,
                memory_before=before, memory_after=after, files=sidecars,
                sql_statements=sorted(set(statements)),
                qualification='Synthetic existing-validator measurement, not P07 enforcement or owner capture.')
        finally:
            if tracemalloc.is_tracing():
                tracemalloc.stop()
            if reader.in_transaction:
                reader.execute('ROLLBACK')
            reader.close()
            writer.close()

    def test_05_existing_whole_journal_validation_and_frozen_scan(self):
        with owned_directory() as root, ExitStack() as resources:
            writer, probe, source_bytes = make_journal(root / 'journal.sqlite3', 2000)
            resources.callback(writer.close)
            reader = passive(root / 'journal.sqlite3')
            resources.callback(reader.close)
            try:
                reader.execute('BEGIN')
                records = journal_store.SqliteExecutionJournalStore._verify(probe, reader)[0]
                reader.execute('ROLLBACK')
                first = scan(records, 2000, 0, 50)
                self.assertEqual(first, ([], 1000, 1000, False))
                reader.execute('BEGIN')
                records = journal_store.SqliteExecutionJournalStore._verify(probe, reader)[0]
                reader.execute('ROLLBACK')
                second = scan(records, 2000, first[1], 50)
                self.assertEqual(second, ([1501], 2000, 1000, True))
                with self.assertRaisesRegex(sqlite3.OperationalError, 'readonly'):
                    reader.execute('INSERT INTO journal_entries VALUES(99999,?,?,?,?)',
                                   ('invalid', 'entry_observed', FIXED_TIME, '{}'))
                # New corrupt row beyond pinned H must fail full-current validation.
                bad = record(2000, 2001)
                bad['result_summary'] = {'invalid': 'synthetic-corruption'}
                writer.execute('INSERT INTO journal_entries VALUES(?,?,?,?,?)',
                    (2001, journal._encode(list(journal._subject_key(bad))),
                     bad['event_kind'], bad['recorded_at'], journal._encode(bad)))
                with self.assertRaises(journal.JournalIntegrityError):
                    journal_store.SqliteExecutionJournalStore._verify(probe, reader)
                self.evidence['frozen_scan'] = dict(high_water=2000,
                    first_page=first, second_page=second, source_bytes=source_bytes,
                    corrupt_after_high_water_rejected=True,
                    limitation='Test-only scanner; no production cursor/context/auth implementation.')
            finally:
                reader.close()
                writer.close()
        examples = [journal.JournalRecord(journal._encode(record(i, i + 1, run_id='same')))
                    for i in range(3)]
        self.assertEqual(scan(examples, 3, 0, 1, run_id='same'), ([1], 1, 1, False))
        self.assertEqual(scan(examples, 3, 1, 1, run_id='same'), ([2], 2, 1, False))
        self.assertEqual(scan([], 0, 0, 50), ([], 0, 0, True))

                reader = passive(path)

                deadline = time.perf_counter_ns() + 20_000_000
                calls = 0
                def progress():
                    nonlocal calls
                    calls += 1

    def test_07_controlled_writer_priority_model(self):
        condition = threading.Condition()
        state = dict(readers=1, waiting=0, writer=False, closing=False)
        queued, entered, release, late_entered = [threading.Event() for _ in range(4)]
        order = []
        errors = []
        def writer():
            try:
                with condition:
                    state['waiting'] += 1
                    queued.set()
                    self.assertTrue(condition.wait_for(lambda: state['readers'] == 0, timeout=2))
                    state['waiting'] -= 1
                    state['writer'] = True
                    order.append('writer')
                    entered.set()
                self.assertTrue(release.wait(2))
                with condition:
                    state['writer'] = False
                    condition.notify_all()
            except BaseException as error:
                errors.append(repr(error))
        def late_reader():
            try:
                with condition:
                    self.assertTrue(condition.wait_for(
                        lambda: not state['waiting'] and not state['writer'] and not state['closing'], timeout=2))
                    order.append('late-reader')
                    late_entered.set()
            except BaseException as error:
                errors.append(repr(error))
        first = threading.Thread(target=writer, daemon=True)
        late = threading.Thread(target=late_reader, daemon=True)
        start = time.perf_counter_ns()
        first.start()
        self.assertTrue(queued.wait(2))
        late.start()
        with condition:
            state['readers'] = 0
            condition.notify_all()
        self.assertTrue(entered.wait(2))
        self.assertFalse(late_entered.is_set())
        release.set()
        first.join(2)
        late.join(2)
        self.assertFalse(first.is_alive() or late.is_alive())
        self.assertEqual(errors, [])
        self.assertEqual(order, ['writer', 'late-reader'])
        self.evidence['writer_priority_model'] = dict(order=order,
            elapsed_ns=time.perf_counter_ns() - start,
            qualification='Controlled scheduling model only; existing store RLock unchanged.')
